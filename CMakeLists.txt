cmake_minimum_required(VERSION 3.14)

# ---- In-source guard ----

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "In-source builds are not allowed. Please create a build directory.")
endif()

project(
    orderbook-exploration
    VERSION 0.1.0
    DESCRIPTION "A limit order book, built several ways"
    HOMEPAGE_URL "https://github.com/jfreun123/orderbook-exploration"
    LANGUAGES CXX
)

# ---- Developer mode ----

option(orderbook-exploration_DEVELOPER_MODE "Enable developer mode" ON)

# ---- Build source ----

add_subdirectory(source)
add_subdirectory(app)

# ---- Developer mode targets ----

if(orderbook-exploration_DEVELOPER_MODE)
  include(cmake/dev-mode.cmake)
endif()
